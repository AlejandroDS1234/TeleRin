TeleRin

// el proyecto utiliza el framework Flask y la biblioteca React para desarrollar la UI. La estructuracion del proyecto es la siguiente: 

>".venv" // entorno virtual de python
>"TeleRin" // carpeta principal del proyecto
>"db-init" // carpeta con la base de datos 
>"Fotos" // carpeta con las fotos de los usuarios y del programa
>"react" // carpeta con los archivos react
    >"src" // archivos para el desarrollo de la interfaz
        >"App.css" // estilos para la aplicacion
        >"App.tsx" // componente principal de la app
        >"README.md" //instrucciones especificas de la carpeta 'react'
        >"index.css" // estilos globales
        >"main.tsx" // punto de entrada de la aplicacion react
>"static" // carpeta con los archivos estaticos
    >"CSS" // carpeta con los archivos CSS (por modificar)
    >"iconos" // carpeta con los iconos utilizados en la aplicacion
    >"JS" // carpeta con los archivos JavaScript (por modificar)
    >"tinymce" // carpeta con los archivos necesarios para el editor de texto TinyMCE
>"templates" // carpeta con los archivos HTML
    >"layouts" // plantilla HTML principal
        >"estructura.html" // plantilla base para las paginas del programa
        >"menu_lateral.html" // plantilla para el menu lateral
    >"pagina" // carpeta con las paginas HTML del programa
    >"index.html" // pagina de inicio del programa
>"app.py" // archivo principal de la aplicacion Flask
>"docker-compose.yml" // archivo de configuracion para Docker Compose
>"Dockerfile" // archivo de configuracion para Docker
>"requirements.txt" // archivo con las dependencias del proyecto
>"README.md" // este archivo de instrucciones generales para el proyecto
